    "## Section 3: Data Cleaning"
   ]
  },
  {
   "cell_type": "code",
   "execution_count": null,
   "metadata": {},
   "outputs": [],
   "source": [
    "# Work on a copy so the raw data is always available for reference\n",
    "df = df_raw.copy()\n",
    "\n",
    "# --- Step 1: Drop the empty artifact column ---\n",
    "df.drop(columns=['Unnamed: 22'], inplace=True)\n",
    "print('Dropped Unnamed: 22')\n",
    "\n",
    "# --- Step 2: Parse Date column to datetime ---\n",
    "# Format is MM-DD-YY  e.g.  04-30-22\n",
    "df['Date'] = pd.to_datetime(df['Date'], format='%m-%d-%y', errors='coerce')\n",
    "print('Date column parsed. Invalid dates set to NaT:', df['Date'].isna().sum())\n",
    "\n",
    "# --- Step 3: Remove rows where Amount is null or zero ---\n",
    "# These are cancelled/unresolved orders with no monetary value\n",
    "before = len(df)\n",
    "df = df.dropna(subset=['Amount'])\n",
    "df = df[df['Amount'] > 0]\n",
    "print(f'Removed {before - len(df)} rows with null/zero Amount. Remaining: {len(df)}')\n",
    "\n",
    "# --- Step 4: Remove duplicate Order IDs ---\n",
    "# Keep the first occurrence\n",
    "before = len(df)\n",
    "df.drop_duplicates(subset=['Order ID'], keep='first', inplace=True)\n",
    "print(f'Removed {before - len(df)} duplicate Order IDs. Remaining: {len(df)}')\n",
    "\n",
    "# --- Step 5: Standardise text columns ---\n",
    "# Strip whitespace and unify casing for geography columns\n",
    "df['ship-state'] = df['ship-state'].str.strip().str.upper()\n",
    "df['ship-city']  = df['ship-city'].str.strip().str.upper()\n",
    "\n",
    "# --- Step 6: Fill missing Courier Status with 'Unknown' ---\n",
    "df['Courier Status'].fillna('Unknown', inplace=True)\n",
    "\n",
    "# --- Step 7: Add derived time columns ---\n",
    "df['Month']      = df['Date'].dt.to_period('M').astype(str)\n",
    "df['Month_Name'] = df['Date'].dt.strftime('%b %Y')\n",
    "df['Week']       = df['Date'].dt.isocalendar().week\n",
    "\n",
    "# --- Step 8: Add a Has_Promotion flag ---\n",
    "df['Has_Promotion'] = df['promotion-ids'].notna().astype(int)\n",
    "\n",
    "print('\\nCleaned Dataset Shape:', df.shape)\n",
    "print('Date range:', df['Date'].min().date(), 'to', df['Date'].max().date())"
